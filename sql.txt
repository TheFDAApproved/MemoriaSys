--  ----------------------------------------------------------------------
--  BLOCKS (sections / areas)
--  ----------------------------------------------------------------------
CREATE TABLE blocks (
    block_id INT AUTO_INCREMENT PRIMARY KEY,
    block_name VARCHAR(100) UNIQUE NOT NULL,
    block_type ENUM(
        'Niche', 'Bone Chamber', 'Lawn/Grounds',
        'Unmapped Area', 'Private', 'Mausoleum',
        'Mass Grave', 'Cluster', 'Block'
    ) NOT NULL,
    image_link TEXT,
    coordinates JSON, -- for mapping visuals in the UI
    remarks TEXT
);

--  ----------------------------------------------------------------------
--  GRAVES (individual slots inside a block)
--      Status is manual: 'Vacant' or 'Occupied'.
--      If a grave is not usable, set status = 'Occupied' and add a remark.
--  ----------------------------------------------------------------------
CREATE TABLE graves (
    grave_id INT AUTO_INCREMENT PRIMARY KEY,
    block_id INT NOT NULL,
    grave_code VARCHAR(50) UNIQUE,
    row_num INT,
    col_num INT,
    status ENUM('Vacant', 'Occupied') DEFAULT 'Vacant',
    remarks TEXT,
    FOREIGN KEY (block_id) REFERENCES blocks(block_id) ON DELETE RESTRICT
);

--  ----------------------------------------------------------------------
--  INTERMENTS (burial records)
--      Pending means that the burial is not yet executed.
--      Inactive means not in the cemetery anymore, maybe moved to family or others in remarks
--      Active means in the cemetery
--      To reserve, fill what is needed, and their status is Pending
--      The current occupant is needed to be find a new grave or location so they will still be active if in the cemetery
--      After the current occupant is transfered, then the pending can be now active
--  ----------------------------------------------------------------------
CREATE TABLE interments (
    interment_id INT AUTO_INCREMENT PRIMARY KEY,

    -- Reference / control
    control_number VARCHAR(100) UNIQUE NOT NULL,

    -- Deceased details
    deceased_name VARCHAR(255) NOT NULL,
    last_known_address TEXT,
    death_certificate VARCHAR(100),
    deceased_date_of_birth DATE,
    deceased_date_of_death DATE,
    deceased_sex ENUM('Male', 'Female', 'Unknown') DEFAULT 'Unknown',

    -- Grave (current location)
    -- Can be null if not yet found or maybe in somewhere in the cemetery but forgotten or something 
    current_grave_id INT,
    transfer_to_grave INT,

    -- Contact person (stored directly)
    contact_person_name VARCHAR(255),
    contact_person_phone_number VARCHAR(50),
    contact_person_email VARCHAR(150),
    contact_person_address TEXT,

    -- Assistance type
    assistance_type ENUM('Burial', 'Transfer the remains of the late', 'Other')
        NOT NULL DEFAULT 'Burial',

    -- Permits
    burial_permit_number VARCHAR(100),
    burial_permit_date DATE,
    transfer_permit_number VARCHAR(100),
    transfer_permit_issued_by VARCHAR(150),
    transfer_permit_date DATE,
    exhumation_permit_number VARCHAR(100),
    exhumation_permit_date DATE,

    -- Important dates
    date_buried DATE,
    date_exhumed DATE,
    burial_clearance_date DATE,
    lease_expiration_date DATE,

    -- Status: Pending → Active → Inactive
    status ENUM('Pending', 'Active', 'Inactive') NOT NULL DEFAULT 'Pending',

    -- Free‑text remarks
    remarks TEXT,

    FOREIGN KEY (current_grave_id) REFERENCES graves(grave_id) ON DELETE RESTRICT,
    FOREIGN KEY (transfer_to_grave) REFERENCES graves(grave_id) ON DELETE RESTRICT
);

-- ----------------------------------------------------------------------
-- TRANSFER LOG – Audit trail of grave movements for each interment
-- ----------------------------------------------------------------------
CREATE TABLE transfer_log (
    log_id INT AUTO_INCREMENT PRIMARY KEY,
    interment_id INT NOT NULL,
    from_grave_id INT NULL,          -- NULL if they had no previous grave (new burial)
    to_grave_id INT NULL,            -- NULL if they were removed from the cemetery
    transfer_date DATETIME NOT NULL, -- ⬅️ Exact date + time of the transfer
    reason TEXT,                     -- captured from interments.remarks at time of move

    FOREIGN KEY (interment_id) REFERENCES interments(interment_id) ON DELETE CASCADE
);

DELIMITER //

CREATE TRIGGER log_grave_transfer
BEFORE UPDATE ON interments
FOR EACH ROW
BEGIN
    -- Only fire if the current_grave_id has actually changed
    IF (OLD.current_grave_id <=> NEW.current_grave_id) = 0 THEN
        
        INSERT INTO transfer_log (
            interment_id,
            from_grave_id,
            to_grave_id,
            transfer_date,    -- ⬅️ Stores the exact moment of the move
            reason
        ) VALUES (
            NEW.interment_id,
            OLD.current_grave_id,
            NEW.current_grave_id,
            NOW(),            -- ⬅️ Exact date + time the trigger ran
            NEW.remarks
        );

    END IF;
END//

DELIMITER ;

-- ----------------------------------------------------------------------
-- INDEXES for blocks
-- ----------------------------------------------------------------------
-- block_name is already UNIQUE (implicit index)
-- No additional indexes needed on blocks currently.

-- ----------------------------------------------------------------------
-- INDEXES for graves
-- ----------------------------------------------------------------------
-- Foreign key to blocks (commonly used in JOINs)
CREATE INDEX idx_graves_block_id ON graves(block_id);

-- Filtering by status (Vacant/Occupied) for reservation queries
CREATE INDEX idx_graves_status ON graves(status);

-- Composite index for quick lookup of vacant graves within a block
-- Useful in reserve.php when filtering by block_id and status
CREATE INDEX idx_graves_block_status ON graves(block_id, status);

-- ----------------------------------------------------------------------
-- INDEXES for interments
-- ----------------------------------------------------------------------
-- Foreign keys (already indexed? InnoDB auto-indexes FK columns, but explicit is safer)
CREATE INDEX idx_interments_current_grave_id ON interments(current_grave_id);
CREATE INDEX idx_interments_transfer_to_grave ON interments(transfer_to_grave);

-- Status is heavily filtered (Pending, Active, Inactive)
CREATE INDEX idx_interments_status ON interments(status);

-- control_number is already UNIQUE (implicit index)

-- For searching by deceased_name (supports prefix searches)
-- If you frequently do LIKE 'name%', this helps; for leading wildcards, consider FULLTEXT.
CREATE INDEX idx_interments_deceased_name ON interments(deceased_name(100));

-- Critical for reserve.php: finding expiring/expired leases
CREATE INDEX idx_interments_lease_expiration ON interments(lease_expiration_date);

-- Composite index for common filtering in records.php: status + current_grave_id
CREATE INDEX idx_interments_status_grave ON interments(status, current_grave_id);

-- Composite index for date range queries (e.g., reports by date_buried)
CREATE INDEX idx_interments_date_buried ON interments(date_buried);

-- ----------------------------------------------------------------------
-- INDEXES for transfer_log
-- ----------------------------------------------------------------------
-- Foreign key to interments (heavily used to fetch history)
-- InnoDB auto-indexes FK columns, but we'll add for clarity and documentation.
CREATE INDEX idx_transfer_log_interment_id ON transfer_log(interment_id);

-- If you often query by transfer_date for reporting
CREATE INDEX idx_transfer_log_transfer_date ON transfer_log(transfer_date);

-- Composite index for timeline ordering (interment_id + transfer_date)
-- Speeds up the history fetch for a given interment
CREATE INDEX idx_transfer_log_interment_date ON transfer_log(interment_id, transfer_date);

CREATE TABLE settings (
    setting_id INT AUTO_INCREMENT PRIMARY KEY,
    setting_key VARCHAR(50) NOT NULL UNIQUE,
    setting_value TEXT,
    description VARCHAR(255) NULL
);

CREATE TABLE users (
    user_id INT PRIMARY KEY AUTO_INCREMENT, 
    username VARCHAR(50) UNIQUE NOT NULL,
    role ENUM('Administrator', 'Office Staff', 'Grounds Staff') NOT NULL DEFAULT 'Grounds Staff',
    status ENUM('Verified', 'Unverified') NOT NULL DEFAULT 'Unverified',
    email VARCHAR(255) UNIQUE NOT NULL,
    phone_number VARCHAR(20) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    name VARCHAR(255) NOT NULL
);

CREATE TABLE payments(
    payment_id INT AUTO_INCREMENT PRIMARY KEY,
    reference_number VARCHAR(100) UNIQUE NOT NULL,
    payment_channel VARCHAR(100) NOT NULL, 
    amount DECIMAL(10,2) NOT NULL,
    deceased_name  VARCHAR(255),
    payers_phone_number VARCHAR(20),
    payers_email VARCHAR(255), 
    payers_name VARCHAR(255),
   
    purpose VARCHAR(100) NOT NULL, 
    image_link TEXT, 
    
    -- Audit & Approvals
    confirmed_office_staff INT NULL,
    confirmed_ground_staff INT NULL,
    remarks_payer TEXT,
    remarks_office TEXT,
    remarks_grounds TEXT,
    
    FOREIGN KEY (confirmed_office_staff) REFERENCES users(user_id) ON DELETE SET NULL,
    FOREIGN KEY (confirmed_ground_staff) REFERENCES users(user_id) ON DELETE SET NULL
);
