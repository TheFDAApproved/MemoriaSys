DROP DATABASE memoria_db;
CREATE DATABASE memoria_db;
USE memoria_db;

--  CEMETERY DATABASE SCHEMA
--
--  Three tables, three tenses:
--
--    interments          PRESENT.  What is true right now.
--                        One row per deceased.  Stable ID.
--
--    reservations        FUTURE.   What is planned to become true.
--                        Frozen after creation.  Only Confirm / Cancel.
--
--    placement_history   PAST.     Physical placements that happened.
--                        Append-only.  Full snapshot before and after
--                        each placement change.

--  Lifecycle
--    RESERVE.php
--      Creates N (interment, status='Pending', no grave) and a reservation
--      with reservation_state='Active'.  O's interments row is NOT touched.
--      No placement event is written — nothing has moved yet.
--
--    MONITOR.php
--      Confirm -> atomic: applies O's staged biography edits, moves O
--                 to destination, activates N in target grave, writes
--                 one history entry for O and one for N,
--                 sets state='Executed'.
--      Cancel  -> atomic: soft-deletes N, sets state='Cancelled'.
--                 O is NEVER touched.

--  Edits are NOT history
--    placement_history records PHYSICAL PLACEMENTS only:
--      Initial  - a body arrives in a grave (NULL -> grave)
--      Move     - a body relocates (grave -> different grave)
--      Removal  - a body leaves its grave (grave -> NULL)
--
--    Pure field edits (name correction, contact update, permit added)
--    do NOT produce history rows.  They only modify the live interments
--    row.

--  Reservations: two homes for two kinds of data
--    Dedicated columns: the reservation's OWN fields — target grave,
--      old occupant's destination and new status, permits for the move,
--      notes, etc.  These have FK or CHECK constraints and belong in
--      real columns.
--
--    old_edited_fields (JSON): staged biography edits to the OLD
--      occupant.  A DIFF — only the fields the user actually changed.
--      Applied to O's interments row on Confirm; discarded on Cancel.
--      Never includes control_number, current_grave_id, status,
--      date_buried, or lease_expiration_date (those are either locked
--      or handled by dedicated columns above).

--  Invariants
--    Only one Active reservation may reference a given target grave.
--    Only one Active reservation may reference a given old occupant.
--    Only one Active reservation may reference a given incoming person.
--    If old_interment_id is NULL, all old_new_* fields are NULL.
--    If old_new_status = 'Inactive', old_new_grave_id must be NULL.
--    old_new_grave_id must never equal target_grave_id.
--    A person's current_grave_id respects chk_status_grave.
--    placement_history is append-only: no updated_at, no deleted_at.


--  USERS (MUST BE FIRST – referenced by all other tables)
CREATE TABLE users (
    user_id INT PRIMARY KEY AUTO_INCREMENT,

    -- for case sensitive usernames
    username VARCHAR(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin NOT NULL,
    role ENUM('Administrator', 'Office Staff', 'Grounds Staff') NOT NULL DEFAULT 'Grounds Staff',
    status ENUM('Verified', 'Unverified') NOT NULL DEFAULT 'Unverified',
    email VARCHAR(255) NOT NULL,
    phone_number VARCHAR(20) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    name VARCHAR(255) NOT NULL,
    session_token TEXT NULL,  -- Stores JSON: {token, ip, user_agent, time} for single-session enforcement

    -- Audit columns
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    created_by INT NULL,
    updated_by INT NULL,

    -- Generated "shadow" columns for soft-delete-aware uniqueness.
    -- They equal the real value when the row is ACTIVE (deleted_at IS NULL),
    -- and become NULL when the row is SOFT-DELETED. Because MySQL allows
    -- many NULLs in a UNIQUE index, only ACTIVE rows must be unique.
    -- You never write to these columns; they update automatically.
    active_username VARCHAR(50)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN username ELSE NULL END) STORED,
    active_email VARCHAR(255)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN email ELSE NULL END) STORED,
    active_phone VARCHAR(20)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN phone_number ELSE NULL END) STORED,

    -- Unique only among ACTIVE rows
    CONSTRAINT uk_active_username UNIQUE (active_username),
    CONSTRAINT uk_active_email    UNIQUE (active_email),
    CONSTRAINT uk_active_phone    UNIQUE (active_phone),

    FOREIGN KEY (created_by) REFERENCES users(user_id) ON DELETE SET NULL,
    FOREIGN KEY (updated_by) REFERENCES users(user_id) ON DELETE SET NULL
);

--  BLOCKS (sections / areas)

--  GRAVES (individual slots inside a block)
--  Status is manual: 'Vacant', 'Occupied', or 'Other'.
--  Grave_code: Block name-rowletter-colnumber
CREATE TABLE graves (
    grave_id INT AUTO_INCREMENT PRIMARY KEY,
    block_id INT NOT NULL,
    grave_code VARCHAR(50) NOT NULL,
    row_num INT,
    col_num INT,
    status ENUM('Vacant', 'Occupied', 'Other') DEFAULT 'Vacant',
    remarks TEXT,

    -- Audit columns
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    created_by INT NULL,
    updated_by INT NULL,

    -- Shadow column: unique only among ACTIVE graves.
    -- When a block is soft-deleted, its graves also get deleted_at set in PHP okay,
    -- so their grave_code becomes reusable for future graves.
    active_grave_code VARCHAR(50)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN grave_code ELSE NULL END) STORED,
    CONSTRAINT uk_active_grave_code UNIQUE (active_grave_code),

    FOREIGN KEY (block_id) REFERENCES blocks(block_id) ON DELETE RESTRICT,
    FOREIGN KEY (created_by) REFERENCES users(user_id) ON DELETE SET NULL,
    FOREIGN KEY (updated_by) REFERENCES users(user_id) ON DELETE SET NULL
);

--  INTERMENTS (PRESENT — current state)
--      Pending  = burial is not yet executed (not physically interred yet).
--      Active   = in the cemetery.
--      Inactive = was interred, now removed (moved to family, exhumed, etc.).
--      Reservations live in the `reservations` table; this table only
--      tracks where the body is *right now* via current_grave_id.
CREATE TABLE interments (
    interment_id INT AUTO_INCREMENT PRIMARY KEY,

    -- Reference / control
    control_number VARCHAR(100) NOT NULL,

    -- Deceased details
    deceased_name VARCHAR(255) NOT NULL,
    last_known_address TEXT,
    death_certificate VARCHAR(100),
    deceased_date_of_birth DATE,
    deceased_date_of_death DATE,
    deceased_sex ENUM('Male', 'Female', 'Unknown') DEFAULT 'Unknown',

    -- Grave (current location)
    -- Can be null if not yet found or maybe in somewhere in the cemetery but forgotten or something
    -- Can be status active and null, this just means its in the cemetery. Inactive really means its not in the cemetery anymore, maybe the family gets them or etc. i guess
    current_grave_id INT NULL,

    -- Contact person
    contact_person_name VARCHAR(255),
    contact_person_phone_number VARCHAR(50),
    contact_person_email VARCHAR(150),
    contact_person_address_barangay TEXT,
    contact_person_address TEXT,

    -- Assistance type
    assistance_type ENUM('Burial', 'Transfer the remains of the late', 'Other')
        NOT NULL DEFAULT 'Burial',

    -- Permits
    burial_permit_number VARCHAR(100),
    burial_permit_date DATE,
    transfer_permit_number VARCHAR(100),
    transfer_permit_issued_by VARCHAR(150),
    transfer_permit_date DATE,
    exhumation_permit_number VARCHAR(100),
    exhumation_permit_date DATE,

    -- Important dates
    date_buried DATE,
    date_exhumed DATE,
    burial_clearance_date DATE,
    lease_expiration_date DATE,

    -- Status: Pending → Active → Inactive
    status ENUM('Pending', 'Active', 'Inactive') NOT NULL DEFAULT 'Pending',

    -- Free-text remarks
    remarks TEXT,

    -- Audit columns
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    created_by INT NULL,
    updated_by INT NULL,

    -- Shadow column: unique only among ACTIVE interments
    active_control_number VARCHAR(100)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN control_number ELSE NULL END) STORED,
    CONSTRAINT uk_active_control_number UNIQUE (active_control_number),

    FOREIGN KEY (current_grave_id) REFERENCES graves(grave_id) ON DELETE RESTRICT,
    FOREIGN KEY (created_by) REFERENCES users(user_id) ON DELETE SET NULL,
    FOREIGN KEY (updated_by) REFERENCES users(user_id) ON DELETE SET NULL,

    -- If status is Pending or Inactive, current_grave_id must be NULL.
    -- Active rows may have NULL (bone chamber / location unknown).
    CONSTRAINT chk_status_grave CHECK (
        status = 'Active' OR current_grave_id IS NULL
    )
);


--  RESERVATIONS (FUTURE — planned data)
--  A reservation is a proposal to place an incoming person (N) into a
--  target grave (G), possibly displacing an existing occupant (O).
--  Reservation lifecycle (reservation_state):
--    Active     Waiting for Confirm or Cancel. Frozen after creation.
--    Executed   Movement applied. Kept forever for audit.
--    Cancelled  N discarded, O untouched. Kept forever for audit.
--  Editability:
--    Everything on the reservation is set once, at creation. Nothing is
--    edited after that. To change a reservation, cancel it and create a
--    new one.
--  Uniqueness:
--    Only one Active reservation may target a given grave, reference a
--    given old occupant, or reference a given incoming person. Enforced
--    with shadow columns keyed on reservation_state = 'Active'.
--  Old-occupant fields (old_new_*):
--    Only set when old_interment_id IS NOT NULL.
--    old_new_grave_id = O's destination grave. NULL means "no grave"
--      (either bone chamber with status='Active', or removed from the
--      cemetery with status='Inactive').
--    old_new_status   = O's new status: Active or Inactive.
--    old_new_grave_id must never equal target_grave_id (no co-interment
--      through a reservation).
--  old_edited_fields (JSON):
--    Staged biography edits to O as a DIFF — only fields the user
--    actually changed.  Applied to O's interments row on Confirm in the
--    same UPDATE as the move.  Discarded on Cancel.
--    Never contains control_number, current_grave_id, status,
--    date_buried, or lease_expiration_date.
CREATE TABLE reservations (
    reservation_id INT AUTO_INCREMENT PRIMARY KEY,

    -- The incoming occupant. Always set.
    pending_interment_id INT NOT NULL,

    -- The grave N will occupy. Must be Vacant, or must hold exactly one
    -- occupant (O) who is leaving.
    target_grave_id INT NOT NULL,

    -- The Active interment being displaced. NULL when target was vacant.
    old_interment_id INT NULL,

    -- Where O goes when the reservation is confirmed.
    old_new_grave_id INT NULL,
    old_new_status ENUM('Active','Inactive') NULL,
    old_new_assistance_type ENUM('Burial','Transfer the remains of the late','Other') NULL,
    old_new_remarks TEXT NULL,

    -- Permits for O's move.
    exhumation_permit_number VARCHAR(100) NULL,
    transfer_permit_number VARCHAR(100) NULL,

    -- Free-text notes about the reservation itself ("why this exists").
    reservation_notes TEXT NULL,

    -- Lifecycle
    reservation_state ENUM('Active','Executed','Cancelled') NOT NULL DEFAULT 'Active',
    actioned_at DATETIME NULL,
    actioned_by INT NULL,

    -- Staged O biography edits (JSON DIFF).  Applied on Confirm; discarded on Cancel.
    old_edited_fields JSON NULL,

    -- Audit
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    created_by INT NULL,
    updated_by INT NULL,

    -- Shadow columns: uniqueness only among Active reservations.
    -- Because MySQL allows multiple NULLs in a UNIQUE index, reservations
    -- in Executed / Cancelled state do not block each other.
    active_pending_interment_id INT
        GENERATED ALWAYS AS (CASE WHEN reservation_state = 'Active' THEN pending_interment_id ELSE NULL END) STORED,
    active_target_grave_id INT
        GENERATED ALWAYS AS (CASE WHEN reservation_state = 'Active' THEN target_grave_id ELSE NULL END) STORED,
    active_old_interment_id INT
        GENERATED ALWAYS AS (CASE WHEN reservation_state = 'Active' THEN old_interment_id ELSE NULL END) STORED,

    CONSTRAINT uk_active_pending_interment UNIQUE (active_pending_interment_id),
    CONSTRAINT uk_active_target_grave      UNIQUE (active_target_grave_id),
    CONSTRAINT uk_active_old_interment     UNIQUE (active_old_interment_id),

    -- Foreign keys
    FOREIGN KEY (pending_interment_id) REFERENCES interments(interment_id) ON DELETE RESTRICT,
    FOREIGN KEY (target_grave_id)      REFERENCES graves(grave_id)          ON DELETE RESTRICT,
    FOREIGN KEY (old_interment_id)     REFERENCES interments(interment_id)  ON DELETE RESTRICT,
    FOREIGN KEY (old_new_grave_id)     REFERENCES graves(grave_id)          ON DELETE RESTRICT,
    FOREIGN KEY (actioned_by)          REFERENCES users(user_id)            ON DELETE SET NULL,
    FOREIGN KEY (created_by)           REFERENCES users(user_id)            ON DELETE SET NULL,
    FOREIGN KEY (updated_by)           REFERENCES users(user_id)            ON DELETE SET NULL,

    -- If O is not set, nothing about O is set. If O is set, its new
    -- status must be decided.
    CONSTRAINT chk_old_pairing CHECK (
        (old_interment_id IS NULL
            AND old_new_grave_id IS NULL
            AND old_new_status IS NULL
            AND old_new_assistance_type IS NULL)
        OR
        (old_interment_id IS NOT NULL AND old_new_status IS NOT NULL)
    ),

    -- An Inactive O has no grave.
    CONSTRAINT chk_inactive_has_no_grave CHECK (
        NOT (old_new_status = 'Inactive' AND old_new_grave_id IS NOT NULL)
    ),

    -- O's destination must differ from the target grave.
    CONSTRAINT chk_old_new_grave_not_target CHECK (
        old_new_grave_id IS NULL OR old_new_grave_id <> target_grave_id
    )
);


--  PLACEMENT_HISTORY (PAST — append-only)
--  Every change of a person's physical placement produces one row here.
--  Field edits (name correction, contact change, permit added) do NOT
--  produce history rows — they only modify the live interments row.
--  Snapshot semantics:
--    snapshot_before  = full JSON copy of the interment row before the
--                       placement change.
--    snapshot_after   = full JSON copy of the interment row after the
--                       placement change.
--  Cross-referencing:
--    related_interment_id  = the other person involved in the same
--                            reservation operation (O ↔ N). NULL for
--                            manual moves.
--    reservation_id        = the reservation that caused the change.
--                            NULL for manual moves.
--  Append-only: no updated_at, no updated_by, no deleted_at. Rows are
--  never edited or removed. Corrections are made by inserting a
--  compensating row.
--  Event types (physical placements only):
--    Initial   from_grave_id IS NULL,     to_grave_id IS NOT NULL
--    Move      from_grave_id IS NOT NULL, to_grave_id IS NOT NULL
--    Removal   from_grave_id IS NOT NULL, to_grave_id IS NULL
CREATE TABLE placement_history (
    history_id INT AUTO_INCREMENT PRIMARY KEY,

    interment_id INT NOT NULL,
    event_type ENUM('Initial','Move','Removal') NOT NULL,

    from_grave_id INT NULL,
    to_grave_id   INT NULL,

    event_date DATETIME NOT NULL,
    reason     TEXT NULL,

    related_interment_id INT NULL,
    reservation_id       INT NULL,

    snapshot_before JSON NULL,
    snapshot_after  JSON NULL,

    -- Append-only audit
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    created_by INT NULL,

    FOREIGN KEY (interment_id)         REFERENCES interments(interment_id)  ON DELETE RESTRICT,
    FOREIGN KEY (from_grave_id)        REFERENCES graves(grave_id)           ON DELETE RESTRICT,
    FOREIGN KEY (to_grave_id)          REFERENCES graves(grave_id)           ON DELETE RESTRICT,
    FOREIGN KEY (related_interment_id) REFERENCES interments(interment_id)   ON DELETE SET NULL,
    FOREIGN KEY (reservation_id)       REFERENCES reservations(reservation_id) ON DELETE SET NULL,
    FOREIGN KEY (created_by)           REFERENCES users(user_id)             ON DELETE SET NULL
);


--  SETTINGS
CREATE TABLE settings (
    setting_id INT AUTO_INCREMENT PRIMARY KEY,
    setting_key VARCHAR(50) NOT NULL,
    setting_value TEXT,
    description VARCHAR(255) NULL,

    -- Audit columns
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    created_by INT NULL,
    updated_by INT NULL,

    -- Shadow column: unique only among ACTIVE settings
    active_setting_key VARCHAR(50)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN setting_key ELSE NULL END) STORED,
    CONSTRAINT uk_active_setting_key UNIQUE (active_setting_key),

    FOREIGN KEY (created_by) REFERENCES users(user_id) ON DELETE SET NULL,
    FOREIGN KEY (updated_by) REFERENCES users(user_id) ON DELETE SET NULL
);

    payment_id INT AUTO_INCREMENT PRIMARY KEY,
    reference_number VARCHAR(100) NOT NULL,
    payment_channel VARCHAR(100) NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    deceased_name  VARCHAR(255),
    payers_phone_number VARCHAR(20),
    payers_email VARCHAR(255),
    payers_name VARCHAR(255),

    purpose VARCHAR(100) NOT NULL,
    image_link TEXT,

    -- Audit & Approvals
    confirmed_office_staff INT NULL,
    confirmed_ground_staff INT NULL,
    remarks_payer TEXT,
    remarks_office TEXT,
    remarks_grounds TEXT,

    -- Audit columns
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at DATETIME NULL,
    created_by INT NULL,
    updated_by INT NULL,

    -- Shadow column: unique only among ACTIVE payments
    active_reference_number VARCHAR(100)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN reference_number ELSE NULL END) STORED,
    CONSTRAINT uk_active_reference_number UNIQUE (active_reference_number),

    FOREIGN KEY (confirmed_office_staff) REFERENCES users(user_id) ON DELETE SET NULL,
    FOREIGN KEY (confirmed_ground_staff) REFERENCES users(user_id) ON DELETE SET NULL,
    FOREIGN KEY (created_by) REFERENCES users(user_id) ON DELETE SET NULL,
    FOREIGN KEY (updated_by) REFERENCES users(user_id) ON DELETE SET NULL
);
